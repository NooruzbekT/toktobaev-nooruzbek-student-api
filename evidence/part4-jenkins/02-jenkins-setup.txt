nooruzbek@NooruzbekT:~/toktobaev-nooruzbek-jenkins$ cat start-jenkins.sh
#!/bin/bash
# Builds and starts the Jenkins controller of Nooruzbek Toktobaev on http://localhost:8090
# First run: create jenkins.env (chmod 600) with JENKINS_ADMIN_PASSWORD=... and GITHUB_TOKEN=...
set -e
cd "$(dirname "$0")"

if [ ! -f jenkins.env ]; then
    echo "jenkins.env is missing (JENKINS_ADMIN_PASSWORD, GITHUB_TOKEN)" >&2
    exit 1
fi

docker build -t toktobaev-nooruzbek-jenkins:lts-jdk21 .

docker rm -f toktobaev-nooruzbek-jenkins >/dev/null 2>&1 || true
# -u root: the jenkins user must be able to use the host Docker socket
docker run -d --name toktobaev-nooruzbek-jenkins \
    -u root \
    -p 8090:8080 \
    -v toktobaev-nooruzbek-jenkins-home:/var/jenkins_home \
    -v /var/run/docker.sock:/var/run/docker.sock \
    -v "$PWD/casc.yaml:/var/jenkins_conf/jenkins.yaml:ro" \
    --env-file jenkins.env \
    -e CASC_JENKINS_CONFIG=/var/jenkins_conf/jenkins.yaml \
    -e JAVA_OPTS="-Djenkins.install.runSetupWizard=false -Xmx768m" \
    -e GRADLE_USER_HOME=/var/jenkins_home/.gradle \
    --memory 3g \
    --restart unless-stopped \
    toktobaev-nooruzbek-jenkins:lts-jdk21

echo "Jenkins is starting on http://localhost:8090 (user: admin)"
